    <script>
        function switchTab(tabId) {
            // Esconde todas as abas
            document.querySelectorAll('.tab-content').forEach(el => {
                el.classList.add('hidden');
            });
            // Mostra a aba selecionada
            document.getElementById('tab-' + tabId).classList.remove('hidden');

            // Remove o estilo ativo de todos os botões
            document.querySelectorAll('.tab-btn').forEach(btn => {
                btn.classList.remove('bg-greenMain', 'text-white', 'shadow-sm');
                btn.classList.add('text-emerald-100');
            });
            // Ativa o botão clicado
            const activeBtn = document.getElementById('btn-' + tabId);
            activeBtn.classList.remove('text-emerald-100');
            activeBtn.classList.add('bg-greenMain', 'text-white', 'shadow-sm');
        }
    </script>
